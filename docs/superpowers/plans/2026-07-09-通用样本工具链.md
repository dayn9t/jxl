# src/jxl/target.py
"""目标类别 profile: 类别定义单一数据源, 驱动 det_mine/doubao_relabel 等 5 处映射."""
import tomllib
from pathlib import Path

from pydantic import BaseModel

_TARGETS_DIR = Path(__file__).parent.parent / "targets"


class TargetProfile(BaseModel):
    """类别 profile: 一处定义, 驱动 YOLOE/GDINO 文本/RF-DETR cls/VLM prompt/权重/输出 cls."""
    name: str
    yolo_text: str
    rfdetr_cls_id: int | None
    vlm_prompt: str
    weights: str
    output_cls_id: int = 0


def load_target(name: str, profile_path: Path | None = None) -> TargetProfile:
    """加载 profile: --target-profile 显式 > targets/<name>.toml 内置 > FileNotFoundError."""
    path = profile_path if profile_path else _TARGETS_DIR / f"{name}.toml"
    if not path.is_file():
        raise FileNotFoundError(f"target profile 不存在: {path}")
    with path.open("rb") as f:
        data = tomllib.load(f)
    return TargetProfile(**data)
```

# src/jxl/det/box_utils.py
"""检测框工具(IoU 等), 供 hardmine / rmb_eval_grounding 共用, 消除重复实现."""

def xyxy_iou(a: tuple[float, float, float, float], b: tuple[float, float, float, float]) -> float:
    """两 xyxy 框 IoU."""
    ax1, ay1, ax2, ay2 = a
    bx1, by1, bx2, by2 = b
    inter_x1, inter_y1 = max(ax1, bx1), max(ay1, by1)
    inter_x2, inter_y2 = min(ax2, bx2), min(ay2, by2)
    iw, ih = max(0.0, inter_x2 - inter_x1), max(0.0, inter_y2 - inter_y1)
    inter = iw * ih
    area_a = (ax2 - ax1) * (ay2 - ay1)
    area_b = (bx2 - bx1) * (by2 - by1)
    union = area_a + area_b - inter
    return inter / union if union > 0 else 0.0
```

修改 `hardmine.py`：删本地 `xyxy_iou`，加 `from jxl.det.box_utils import xyxy_iou`（保留原签名，调用处不变）。
